\section{Contexte et objectifs}

% ── Contexte / problématique ────────────────────────────────────────────────
\begin{frame}{Contexte et problématique}
  \textbf{Classer des candidatures} : plusieurs critères, plusieurs évaluateurs.
  \begin{itemize}
    \item Difficile à rendre \alert{objectif, transparent et sans ambiguïté}.
    \item \alert{Chronophage} : traiter manuellement un grand volume de candidatures.
    \item Critères \alert{variables d'une formation à l'autre}.
  \end{itemize}
  \vspace{0.8em}
  \begin{block}{Problématique}
    Comment \alert{automatiser} ce classement de façon objective, transparente et explicable,
    et qui s'améliore avec les décisions passées ?
  \end{block}
\end{frame}

% ── Objectif ─────────────────────────────────────────────────────────────────
\begin{frame}{Objectif}
  \centering
  Concevoir un \textbf{outil générique} pour automatiser le classement de candidatures,
  de manière transparente et explicable.
  \vspace{1.0em}

  \begin{tikzpicture}[
    p/.style={rectangle, rounded corners, draw=primary, fill=softgray, thick,
              text width=3.3cm, align=center, inner sep=6pt, minimum height=2.4cm}
  ]
    \node[p] (g) {\textbf{Générique}\\[4pt]\footnotesize critères et classes configurables};
    \node[p, right=0.5cm of g] (e) {\textbf{Explicable}\\[4pt]\footnotesize chaque rang justifié};
    \node[p, right=0.5cm of e] (a) {\textbf{Adaptatif}\\[4pt]\footnotesize s'améliore avec les décisions validées};
  \end{tikzpicture}
\end{frame}
