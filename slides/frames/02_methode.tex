% ── AHP ─────────────────────────────────────────────────────────────────────
\begin{frame}{AHP : pondération des critères}
  À partir d'une matrice de comparaison par paires $A=(a_{ij})$ (échelle de Saaty,
  $a_{ji}=1/a_{ij}$), les poids sont le \alert{vecteur propre principal} :
  \[
    A\,w = \lambda_{\max}\,w, \qquad \sum_i w_i = 1.
  \]
  La cohérence des jugements est contrôlée par le ratio de cohérence :
  \[
    CI = \frac{\lambda_{\max}-n}{n-1}, \qquad CR = \frac{CI}{RI}, \qquad
    \text{(valide si } CR < 0{,}10 \text{)}.
  \]
  \begin{block}{Idée}
    Le décideur exprime ses \emph{préférences relatives} ; l'AHP en déduit des poids
    cohérents, ou signale une incohérence à corriger.
  \end{block}
\end{frame}

% ── TOPSIS ──────────────────────────────────────────────────────────────────
\begin{frame}{TOPSIS : classement des candidats}
  Matrice pondérée normalisée, puis distances aux solutions idéales :
  \[
    v_{ij} = w_j \cdot \frac{x_{ij}}{\sqrt{\sum_k x_{kj}^2}}, \qquad
    v_j^{+}=\max_i v_{ij}, \quad v_j^{-}=\min_i v_{ij}.
  \]
  \[
    S_i^{\pm} = \sqrt{\textstyle\sum_j (v_{ij}-v_j^{\pm})^2}, \qquad
    C_i = \frac{S_i^{-}}{S_i^{+}+S_i^{-}} \in [0,1].
  \]
  Les candidats sont classés par $C_i$ décroissant (tous critères en bénéfice :
  valeur haute = meilleur profil).
  \vspace{0.4em}
  \begin{block}{Transparence}
    Chaque rang est expliqué par la contribution de chaque critère au score.
  \end{block}
\end{frame}
